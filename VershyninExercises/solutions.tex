\begin{exercise}
	(Two variance formulas)
\end{exercise}

\begin{solution}
	\begin{enumerate}
		\item 
		\[
		\E \| Z - \E Z \|_2^2 = \E \sum_{i=1}^{n}( Z_i - \E Z_i )^2 
		=  \sum_{i=1}^{n} \E ( Z_i - \E Z_i )^2 = \sum_{i=1}^{n} \E Z_i^2 - (\E Z_i)^2
		\]
		\[
		= \E \sum_{i=1}^{n} Z_i^2 - \sum_{i=1}^{n}  (\E Z_i)^2 = \E \|Z\|_2^2 - \| \E Z \|_2^2 
		\]
		
		Other way:
		
		\[
		\E \|Z-\E Z\|_2^2 = \E (Z-\E Z)^\top (Z-\E Z) = \E Z^\top Z - \E( Z^\top \E Z )- \E ((\E Z)^\top Z) +\E (\E Z)^\top (\E Z)
		\]
		\[
		= \E\|Z\|_2^2 - 2 (\E Z)^\top (\E Z) + (\E Z)^\top (\E Z) = \E \|Z\|_2^2 - \| \E Z \|_2^2 
		\]
		
		\item 
		
		\[
		\E\|Z - \E Z\|_2^2 = \sum_{i=1}^{n} \E (Z_i - \E Z_i )^2 
		=  \sum_{i=1}^{n} \E Z_i^2 - (\E Z_i)^2 = \frac{1}{2} \sum_{i=1}^{n} 2\E Z_i^2 - 2(\E Z_i)^2 
		\]
		\[
		= \frac{1}{2} \sum_{i=1}^{n} \E Z_i^2 + \E (Z_i')^2 - 2(\E Z_i)(\E Z_i')
		\]
		 \[
		 =  \frac{1}{2} \sum_{i=1}^{n} \E Z_i^2 + \E (Z_i')^2 - 2(\E Z_i Z_i') \quad \text{from independence}
		 \]
		 \[
		 = \frac{1}{2} \sum_{i=1}^{n} \E( Z_i^2 + (Z_i')^2 - 2Z_i Z_i') =  \frac{1}{2} \sum_{i=1}^{n} \E (Z_i - Z_i')^2 = 
		 \frac{1}{2} \E\| Z - Z' \|_2^2
		 \]
		
	\end{enumerate}
	
\end{solution}

\begin{exercise}
	
	(Expectation minimizes the mean squared error)
	The question has a typo and it should say:
	\[
	\Var(X) = \min_{a\in\R}\E(X-a)^2
	\]
	
\end{exercise}

\begin{solution}
	First lets check the scalar version:
	\[
	\min_{a\in\R}\E(X-a)^2 = \min_{a\in\R} \E X^2 - 2a \E X + a^2 = \min_{a\in\R} \Var(X) + ((\E X) - a )^2 = \Var(X)
	\]
	
	Now lets prove the euclidean vector version:
	\[
	\min_{a\in\R^n} \E\|Z - a\|_2^2 = \min_{a\in\R^n} \E \|Z\|_2^2 - 2 \langle \E Z, a \rangle + \|a\|_2^2
	\]
	\[
	= \min_{a\in\R^n} \E\|Z-\E Z\|_2^2 + \|\E Z\|_2^2 - 2 \langle \E Z, a \rangle + \|a\|_2^2 \quad \text{from Exercise }0.1
	\]

\end{solution}

\begin{exercise}
	(Variance of a sum) 
\end{exercise}

\begin{solution}
	
	
	\[
	\E\|\sum_{j=1}^{k} Z_j\|_2^2 = \E\sum_{i,j=1}^{k} Z_i^\top Z_j = \E(\sum_{j=1}^{k} \|Z_j\|_2^2 + \sum_{i\ne j} Z_i^\top Z_j) = \sum_{j=1}^{k} \E\|Z_j\|_2^2 + \sum_{i\ne j} \E Z_i^\top Z_j
	\]
	
	where \(Z_i^\top Z_j\) is also a linear combination so:
	\[
	\E Z_i^\top Z_j = (\E Z_i)^\top \E Z_j = 0^\top0 = 0
	\]
\end{solution}

\begin{exercise}
	(Gaussian tails: a lower bound)
\end{exercise}

\begin{solution}
	
	\[
	\Prob(g \geq t) = \int_{t}^{\infty} \frac{1}{\sqrt{2\pi}} e^{-x^2/2}\; dx
	, \quad \frac{d}{dt} \Prob(g \geq t) = - \frac{1}{\sqrt{2\pi}} e^{-t^2/2}
	\]
	\[
	 \frac{d}{dt}  \frac{t}{t^2 + 1} \cdot \frac{1}{\sqrt{2\pi}} e^{-t^2/2}
	= \frac{1}{\sqrt{2\pi}} \big( \frac{1 - 2t^2 - t^4}{(t^2 + 1)^2} \big)  e^{-t^2/2}
	\]
	
defining \(h(t) := \Prob(g \geq t) - \frac{t}{t^2 + 1} \cdot \frac{1}{\sqrt{2\pi}} e^{-t^2/2}\) we can see that 
\[
\frac{d}{dt} h(t) = \frac{-2}{ (t^2+1)^2 \sqrt{2\pi}} e^{-t^2/2}
\].
We have \(h'(t) < 0 \) for all \(t\). and \(\lim_{t\to \infty} h'(t) = 0\). So the function \(h(t)\) is strictly decreasing. We can also verify that \(h(0) > 0\) and \(\lim_{t \to \infty} h(t) = 0\), so the function cannot become negative at any point \(t\).
	
\end{solution}

\begin{exercise}
	(Hoeffding lemma)
\end{exercise}	
\begin{solution}
	\begin{enumerate}
		\item 
		For a r.v. \(X\) we have the random variable \(Z := X - \E X  \in [a',b']\), where \(a' = a - \E X\) and \(b' = b - \E X\). Now \(b'-a' = b - a\) and we only need to prove the lemma for \(Z\) while \(\E Z = 0\).
		
		For any \(C = b - a > 0\) we can define \(Z := \frac{X}{C} \) and so \(Z \in [a', b']\) with \(b'-a' = 1\). Now if the lemma holds for \(Z\) then we have:
		 \[
		 \E e^{\lambda(Z - \E Z)} \leq \exp( \lambda^2 (b' - a')^2 / 8) \implies
		 \E e^{\frac{\lambda}{C}(X - \E X)} \leq \exp( \frac{\lambda^2}{C^2} (b - a)^2 / 8)
		 \],
		 which holds for all \(\lambda' := \lambda/C \in \R\).
		 
		 Now assuming \(X\) and \(Z\) both have zero mean and \(b-a = 1\),  we know from last exercise:
		 \[
		 \E e^{\lambda X} \leq \E e^{\lambda Z}
		 \]
		 and if the result holds for \(Z\) then \(\E e^{\lambda Z} \leq \exp( \lambda^2 (b - a)^2 / 8) \). The RHS does not directly depend on the intervals intermediate values.
		 
		 
		 \item 
		 \[
		 K(\lambda) := \log \E e^{\lambda X}, \quad X \in \{a, b\}
		 \]
		 \[
		 \implies K(0) = \log 1 = 0
		 \]
		 also in this case:
		 \[
		 K'(0)=\left.\frac{\mathbb{E}[X e^{\lambda X}]}{\mathbb{E}[e^{\lambda X}]}\right|_{\lambda=0}
		 = \frac{\E X \cdot 1}{ \E 1 } = \E X
		 \]
		 
		 and
		 
		 \[
		 K'' (\lambda) = \frac{(\E X^2 e^{\lambda X})(\E e^{\lambda X}) - (\E X e^{\lambda X})^2 }{ (\E e^{\lambda X})^2 } 
		 \]
		 \[
		  = \frac{ ( p a^2 e^{\lambda a} +  (1-p) b^2 e^{\lambda b}) (p e^{\lambda a} +  (1-p) e^{\lambda b}) - (p a e^{\lambda a} +  (1-p) b e^{\lambda b})^2 }{ (p e^{\lambda a} +  (1-p) e^{\lambda b})^2 }
		 \]
		 \[
		 = \frac{ p(1-p) e^{\lambda (a+b)}(a^2 + b^2) - 2 p(1-p)ab e^{\lambda (a + b)} }{  (p e^{\lambda a} +  (1-p) e^{\lambda b})^2 }
		 \]
		 
		 From \(\E X = 0\) we have \(pa + (1-p)b = 0 \implies p = \frac{b}{b-a} = b\). So:
		 
		 \[
		 K'' (\lambda) = \frac{ -ab e^{\lambda (a+b)}(a^2 + b^2) + 2 a^2b^2 e^{\lambda (a + b)} }{  (b e^{\lambda a} +  -a e^{\lambda b})^2 }
		 \]
		 \[
		 =  \frac{ -ab e^{\lambda (a+b)} \overbrace{((a^2 + b^2) -2ab)}^{ = (b-a)^2} }{  (b e^{\lambda a} +  -a e^{\lambda b})^2 }
		  = \frac{-ab e^{\lambda (a+b) }}{  (b e^{\lambda a} +  -a e^{\lambda b})^2 } \leq \frac{1}{4}
		 \]
		, where the last inequality is from \(( A + B)^2 \leq 2 AB \).
		 
		 
		 Now we see use Taylor's theorem with Lagrange form of the remainder:
		 \[
		 K(\lambda) = K(0) + \lambda K'(0) + \frac{\lambda^2}{2!}K''(\xi), \quad \forall \xi \in (0, \lambda) \text{ or } (\lambda, 0)
		 \]
	\[
	\implies  K(\lambda) \leq \frac{\lambda^2}{8} \implies \E e^X \leq e^\frac{\lambda^2}{8}
	\]
	
	Also from \(\E X = 0\) we know \(a \leq 0 \leq b\) so \(K''(\lambda) \geq 0\) and therefore \(K(0)\) is at a local minimum and we have a convexity here.
	
	\end{enumerate}
\end{solution}

\begin{solution}
	
	We have:
	\[

	\[
	=  e^{-\lambda t} \prod_{i=1}^{N} \E e^{\lambda( X_i - \E X_i)} 
	\leq e^{-\lambda t} e^{\frac{ \lambda^2 \sum_{i=1}^{N} (b_i-a_i)^2}{8}} =  e^{\frac{ \lambda^2 \sum_{i=1}^{N} (b_i-a_i)^2}{8} -\lambda t} 
	\]
	
	now we choose \(\lambda\) to minimize the bound:
	\[
	\frac{d}{d \lambda} \frac{ \lambda^2 \sum_{i=1}^{N} (b_i-a_i)^2}{8} -\lambda t = \frac{\sum_{i=1}^{N} (b_i-a_i)^2}{4} \lambda - t \implies \lambda^* = \frac{4t}{\sum_{i=1}^{N} (b_i-a_i)^2}
	\]
	
	\[
	\implies \frac{\sum_{i=1}^{N} (b_i-a_i)^2}{4} \lambda^* - t = 0 
	\implies \frac{\sum_{i=1}^{N} (b_i-a_i)^2}{4} (\lambda^*)^2 - t \lambda^* = 0
	\]
	\[
	= \frac{\sum_{i=1}^{N} (b_i-a_i)^2}{8} (\lambda^*)^2 + \frac{\sum_{i=1}^{N} (b_i-a_i)^2}{8} (\lambda^*)^2 - t \lambda^*
	\]
	\[
	\implies  \frac{\sum_{i=1}^{N} (b_i-a_i)^2}{8} (\lambda^*)^2 - t \lambda^* =  - \frac{\sum_{i=1}^{N} (b_i-a_i)^2}{8} (\lambda^*)^2 = -\frac{2t^2}{ \sum_{i=1}^{N} (b_i-a_i)^2 }
	\]
	
\end{solution}

\begin{exercise}
	(Chernoff inequality: left tails)
\end{exercise}
	
	
\begin{solution}
	For \(\lambda \geq 0\) and \(0 < t \leq \mu\)
	\[
	\Prob(S_N \leq t) = \Prob(e^{-\lambda S_N}\geq e^{-\lambda t}) \leq e^{\lambda t} \E e^{-\lambda S_N}
	\]
	\[
	= e^{\lambda t} \prod_{i=1}^{N} ( p_i e^{-\lambda} + (1-p_i)e^0 ) = 
	 e^{\lambda t} \prod_{i=1}^{N} (1 + p_i (e^{-\lambda} - 1) )
	\]
	\[
	\leq e^{\lambda t} \prod_{i=1}^{N} \exp(p_i (e^{-\lambda} - 1)) = e^{\lambda t} \exp( (e^{-\lambda} - 1) \sum_{i=1}^{N} p_i ) = 
	e^{\lambda t} \exp( (e^{-\lambda} - 1) \mu )
	\]
	\[
	= \exp( (e^{-\lambda} - 1) \mu + \lambda t )
	\].
	
	Now we minimize \(K(\lambda) \coloneq  (e^{-\lambda} - 1) \mu + \lambda t \) over \(\lambda \geq 0\):
	\[
	K'(\lambda) =  -\mu e^{-\lambda} + t \implies \lambda^* = - \ln \frac{t}{\mu} \text{  which is non-negative by assumption } 0 < t \leq \mu
	\]
	\[
	K(\lambda^*) = (\frac{t}{\mu} - 1 )\mu - \ln (\frac{t}{\mu})^t = t - \mu - \ln (\frac{t}{\mu})^t
	\]
	\[
	\implies \Prob(S_N \leq t) \leq e^{-\mu} (\frac{e \mu}{t})^t
	\]
\end{solution}

\begin{exercise}
	(Examples of subgaussian distributions)
\end{exercise}

\begin{solution}
	\begin{enumerate}
		\item (Constant).
		\[
		\E e^{X^2/K^2} \leq 2 \implies \int_\mathcal{C} e^{c^2/K^2} f_X(x) \; dx \leq 2
		\]
		\[
		\implies  e^{c^2/K^2} \leq 2 \implies K^2 \geq c^2 / \ln2
		\]
		
		\item (Bounded). First we clarify: \(X\) is bounded a.s. means that \(\|X\|_\infty = \sup_x \{\Prob(X \leq x) = 1\} \) exists. Therefore:
		\[
		 \frac{X^2}{K^2} \leq  \frac{\|X\|_\infty^2}{K^2} \;\text{ a.s.}
		\]
		\[
		\implies \E e^\frac{X^2}{K^2} \leq  e^\frac{\|X\|_\infty^2}{K^2}
		\]
		So for a \(K\) that \(e^\frac{\|X\|_\infty^2}{K^2} \leq 2\) we will definitely have \(\E e^\frac{X^2}{K^2} \leq 2\) and so \(\|X\|_{\subg} \leq K\). So we find this bound for such \(K\):
		\[
		e^\frac{\|X\|_\infty^2}{K^2} \leq 2 \implies K \geq \|X\|_\infty/\sqrt{\ln2}
		\],
		by taking \(K = \|X\|_\infty/\sqrt{\ln2}\) and that \(\|X\|_{\subg} \leq K\), we get \( \|X\|_{\subg} \leq \|X\|_\infty/\sqrt{\ln2} \)
		
		\item (Rademacher).
		\[
		\E e^{\frac{X^2}{K^2}} \leq 2 \implies e^{\frac{1}{K^2}} \leq 2
		\]
		\[
		\implies K \geq \frac{1}{\sqrt{\ln 2}} \implies \|X\|_\subg = \frac{1}{\sqrt{\ln 2}}
		\]
		\item  (Normal) we have \(X \sim N(0, \sigma^2)\)
		\[
		\E e^{\frac{X^2}{K^2}} \leq 2 \implies \frac{1}{\sqrt{2\pi}} \int_{-\infty}^\infty e^{ \frac{(2 - K^2)x^2}{2K^2} } \; dx \leq 2
		\]
		
		Setting \( \frac{(2 - K^2)x^2}{2K^2} = -\frac{t^2}{2} \implies \frac{(K^2-2)}{K^2} x^2 = t^2 \implies dt = \sqrt{\frac{(K^2-2)}{K^2}} \; dx\), we get:
		\[
		  \sqrt\frac{K^2}{K^2-2} \int_{-\infty}^\infty \frac{1}{\sqrt{2\pi}} e^{ \frac{-t^2}{2} } \; dt \leq 2 \implies  \sqrt\frac{K^2}{K^2-2} \leq 2
		\]
		\[
		3K^2-8 \geq 0 \implies K \geq \sqrt{8/3} \text{ or } K \leq -\sqrt{8/3}
		\]
		\item (Bernoulli) 
		
		\[
		\E e^{X^2/K^2} = pe^{1/K^2} + 1-p \leq 2
		\]
		\[
		\implies e^{1/K^2} \leq \frac{1}{p} + 1 \implies K^2 \geq \frac{1}{\sqrt{\ln( \frac{1}{p} + 1 )}}
		\]
		
	\end{enumerate}
\end{solution}
